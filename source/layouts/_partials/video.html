{{- /* click to load youtube facade the poster frame is fetched at build time and re-encoded locally, so the page makes no third party request until the visitor actually presses play params: id (required), title, caption, start */ -}} {{- $id := .id -}} {{- $label := .title | default "" -}} {{- $caption := .caption | default "" -}} {{- $start := .start | default "" -}} {{- if $id -}} {{- $poster := "" -}} {{- range slice "maxresdefault" "hqdefault" -}} {{- if not $poster -}} {{- $url := printf "https://i.ytimg.com/vi/%s/%s.jpg" $id . -}} {{- with try (resources.GetRemote $url) -}} {{- if not .Err -}} {{- with .Value -}} {{- with try (.Resize "960x webp q80") -}} {{- if not .Err -}} {{- $target := printf "media/video/%s.webp" $id -}} {{- $poster = (resources.Copy $target .Value).RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
